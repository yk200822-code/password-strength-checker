import re def check_password(password): score = 0 if len(password) >= 8: score += 1 if re.search(r"[A-Z]", password): score += 1 if re.search(r"[a-z]", password): score += 1 if re.search(r"[0-9]", password): score += 1 if re.search(r"[^A-Za-z0-9]", password): score += 1 print("\n----- PASSWORD REPORT -----") if score <= 2: print("Password Strength: WEAK") print("Suggestion: Create a stronger password.") elif score <= 4: print("Password Strength: MEDIUM") print("Suggestion: Add more security features.") else: print("Password Strength: STRONG") print("Your password meets all five checks!") print("Score:", score, "/ 5") print("PASSWORD STRENGTH CHECKER") password = input("Enter your password: ") check_password(password)
